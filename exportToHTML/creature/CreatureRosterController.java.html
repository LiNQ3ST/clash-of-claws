package creature;

import javafx.fxml.FXML;
import javafx.scene.control.Label;
import javafx.scene.control.ListView;
import javafx.scene.control.TextField;

import java.nio.file.Files;
import java.nio.file.Path;
import java.util.List;

/**
 * Minimal milestone scene. The team's scene factory only needs to load the
 * matching FXML file; generation and persistence stay in the backend classes.
 */
public final class CreatureRosterController {
    @FXML private Label nameLabel;
    @FXML private Label typeLabel;
    @FXML private Label hpLabel;
    @FXML private Label playerCatLabel;
    @FXML private Label statusLabel;
    @FXML private ListView<String> abilityListView;
    @FXML private ListView<Cat> catDexListView;
    @FXML private TextField renameField;

    private CatDAO catDAO;
    private CatGenerator catGenerator;

    @FXML
    private void initialize() {
        String defaultPath = Path.of("data", "clash-of-cats.db").toString();
        String databasePath = System.getProperty("clashofcats.db", defaultPath);
        createParentDirectory(databasePath);

        catDAO = new CatDAO("jdbc:sqlite:" + databasePath);
        catDAO.initializeSchema();
        catGenerator = new CatGenerator();

        catDexListView.getSelectionModel()
                .selectedItemProperty()
                .addListener((observable, oldCat, newCat) -> displayCat(newCat));

        refreshCatDex();
    }

    @FXML
    private void handleGenerateCat() {
        runSafely(() -> {
            Cat cat = catGenerator.generateUniqueAndStore(catDAO);
            refreshCatDex();
            selectById(cat.getId());
            displayCat(cat);
            statusLabel.setText("Generated and stored cat #" + cat.getId());
        });
    }

    @FXML
    private void handleRefresh() {
        runSafely(this::refreshCatDex);
    }

    @FXML
    private void handleRenameSelected() {
        runSafely(() -> {
            Cat selected = catDexListView.getSelectionModel().getSelectedItem();
            if (selected == null) {
                statusLabel.setText("Select a cat before updating");
                return;
            }
            if (renameField.getText() == null || renameField.getText().isBlank()) {
                statusLabel.setText("Enter a new name first");
                return;
            }

            selected.setName(renameField.getText());
            boolean updated = catDAO.update(selected);
            refreshCatDex();
            selectById(selected.getId());
            statusLabel.setText(updated ? "Updated cat #" + selected.getId() : "Cat was not found");
            renameField.clear();
        });
    }

    @FXML
    private void handleDeleteSelected() {
        runSafely(() -> {
            Cat selected = catDexListView.getSelectionModel().getSelectedItem();
            if (selected == null) {
                statusLabel.setText("Select a cat before deleting");
                return;
            }

            boolean deleted = catDAO.delete(selected.getId());
            refreshCatDex();
            clearDetails();
            statusLabel.setText(deleted ? "Deleted cat #" + selected.getId() : "Cat was not found");
        });
    }

    private void refreshCatDex() {
        List<Cat> cats = catDAO.findAll();
        catDexListView.getItems().setAll(cats);
        statusLabel.setText(cats.isEmpty() ? "No cat data" : "Loaded " + cats.size() + " CatDex entries");
    }

    private void displayCat(Cat cat) {
        if (cat == null) {
            clearDetails();
            return;
        }

        nameLabel.setText(cat.getName());
        typeLabel.setText(cat.getType());
        hpLabel.setText(Integer.toString(cat.getHp()));
        playerCatLabel.setText(cat.isPlayerCat() ? "Player cat" : "Generated opponent");
        abilityListView.getItems().setAll(cat.getAbilityIds());
    }

    private void clearDetails() {
        nameLabel.setText("-");
        typeLabel.setText("-");
        hpLabel.setText("-");
        playerCatLabel.setText("-");
        abilityListView.getItems().clear();
    }

    private void selectById(int catId) {
        catDexListView.getItems().stream()
                .filter(cat -> cat.getId() == catId)
                .findFirst()
                .ifPresent(cat -> catDexListView.getSelectionModel().select(cat));
    }

    private void runSafely(Runnable action) {
        try {
            action.run();
        } catch (RuntimeException exception) {
            statusLabel.setText(exception.getMessage());
        }
    }

    private static void createParentDirectory(String databasePath) {
        try {
            Path path = Path.of(databasePath).toAbsolutePath();
            Path parent = path.getParent();
            if (parent != null) {
                Files.createDirectories(parent);
            }
        } catch (Exception exception) {
            throw new IllegalStateException("Could not create the database directory", exception);
        }
    }
}

CatGenerator.java 
package creature;

import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStream;
import java.io.InputStreamReader;
import java.math.BigInteger;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Objects;
import java.util.Random;

/**
 * Backend portion of the CreatureGen slice.
 *
 * It chooses a name, breed/type, type-appropriate HP, and 1-4 unique ability
 * IDs. The arena is responsible for interpreting what each ability ID does.
 */
public final class CatGenerator {
    private static final int MAX_RANDOM_ATTEMPTS = 10_000;

    private final List<String> names;
    private final List<String> abilityIds;
    private final List<CatTypeDefinition> types;
    private final Random random;

    public CatGenerator() {
        this(
                loadResourceLines("/com/clashofcats/creaturegen/data/cat_names.txt"),
                loadResourceLines("/com/clashofcats/creaturegen/data/ability_ids.txt"),
                defaultTypes(),
                new Random()
        );
    }

    public CatGenerator(
            List<String> names,
            List<String> abilityIds,
            List<CatTypeDefinition> types,
            Random random
    ) {
        if (names == null || names.isEmpty()) {
            throw new IllegalArgumentException("At least one cat name is required");
        }
        if (abilityIds == null || abilityIds.isEmpty()) {
            throw new IllegalArgumentException("At least one ability ID is required");
        }
        if (types == null || types.isEmpty()) {
            throw new IllegalArgumentException("At least one cat type is required");
        }

        this.names = List.copyOf(names);
        this.abilityIds = List.copyOf(abilityIds);
        this.types = List.copyOf(types);
        this.random = Objects.requireNonNull(random, "random cannot be null");
    }

    public Cat generateRandomCat() {
        String name = randomItem(names);
        CatTypeDefinition type = randomItem(types);
        int hp = randomBetween(type.minimumHp(), type.maximumHp());

        int largestAbilitySet = Math.min(4, abilityIds.size());
        int abilityCount = randomBetween(1, largestAbilitySet);
        List<String> selectedAbilities = chooseUniqueAbilities(abilityCount);

        return new Cat(name, type.name(), hp, selectedAbilities, false);
    }

    /**
     * Generates an opponent, prevents exact duplicates, inserts it, and returns it.
     */
    public Cat generateUniqueAndStore(CatDAO catDAO) {
        Objects.requireNonNull(catDAO, "catDAO cannot be null");

        BigInteger possible = totalPossibleOpponentCats();
        if (possible.compareTo(BigInteger.valueOf(catDAO.countOpponentCats())) <= 0) {
            throw new IllegalStateException("You already have defeated all the cats");
        }

        for (int attempt = 0; attempt < MAX_RANDOM_ATTEMPTS; attempt++) {
            Cat cat = generateRandomCat();
            if (!catDAO.exists(cat)) {
                return catDAO.insert(cat);
            }
        }

        throw new IllegalStateException(
                "A unique cat could not be found after " + MAX_RANDOM_ATTEMPTS + " attempts"
        );
    }

    /**
     * Stores a user-created player cat through the same CatDex table.
     */
    public Cat createPlayerCatAndStore(
            String name,
            String type,
            int hp,
            List<String> selectedAbilityIds,
            CatDAO catDAO
    ) {
        Objects.requireNonNull(catDAO, "catDAO cannot be null");
        Cat cat = new Cat(name, type, hp, selectedAbilityIds, true);
        return catDAO.insert(cat);
    }

    public BigInteger totalPossibleOpponentCats() {
        BigInteger abilityCombinations = BigInteger.ZERO;
        int largestAbilitySet = Math.min(4, abilityIds.size());

        for (int count = 1; count <= largestAbilitySet; count++) {
            abilityCombinations = abilityCombinations.add(combination(abilityIds.size(), count));
        }

        long hpAndTypeCombinations = types.stream()
                .mapToLong(type -> (long) type.maximumHp() - type.minimumHp() + 1L)
                .sum();

        return BigInteger.valueOf(names.size())
                .multiply(BigInteger.valueOf(hpAndTypeCombinations))
                .multiply(abilityCombinations);
    }

    private List<String> chooseUniqueAbilities(int amount) {
        List<String> shuffled = new ArrayList<>(abilityIds);
        Collections.shuffle(shuffled, random);
        return List.copyOf(shuffled.subList(0, amount));
    }

    private <T> T randomItem(List<T> values) {
        return values.get(random.nextInt(values.size()));
    }

    private int randomBetween(int minimum, int maximum) {
        return random.nextInt(maximum - minimum + 1) + minimum;
    }

    private static BigInteger combination(int n, int k) {
        int smallerK = Math.min(k, n - k);
        BigInteger result = BigInteger.ONE;

        for (int i = 1; i <= smallerK; i++) {
            result = result
                    .multiply(BigInteger.valueOf(n - smallerK + i))
                    .divide(BigInteger.valueOf(i));
        }
        return result;
    }

    private static List<String> loadResourceLines(String resourcePath) {
        InputStream stream = CatGenerator.class.getResourceAsStream(resourcePath);
        if (stream == null) {
            throw new IllegalStateException("Resource not found: " + resourcePath);
        }

        try (BufferedReader reader = new BufferedReader(
                new InputStreamReader(stream, StandardCharsets.UTF_8)
        )) {
            List<String> lines = reader.lines()
                    .map(String::trim)
                    .filter(line -> !line.isBlank())
                    .filter(line -> !line.startsWith("#"))
                    .toList();

            if (lines.isEmpty()) {
                throw new IllegalStateException("Resource contained no usable values: " + resourcePath);
            }
            return lines;
        } catch (IOException exception) {
            throw new IllegalStateException("Could not load resource: " + resourcePath, exception);
        }
    }

    private static List<CatTypeDefinition> defaultTypes() {
        return List.of(
                new CatTypeDefinition("Tabby", 80, 110),
                new CatTypeDefinition("Sphynx", 65, 95),
                new CatTypeDefinition("Maine Coon", 110, 150),
                new CatTypeDefinition("Siamese", 75, 105),
                new CatTypeDefinition("Calico", 85, 120)
        );
    }

    public record CatTypeDefinition(String name, int minimumHp, int maximumHp) {
        public CatTypeDefinition {
            if (name == null || name.isBlank()) {
                throw new IllegalArgumentException("Type name cannot be blank");
            }
            if (minimumHp <= 0 || maximumHp < minimumHp) {
                throw new IllegalArgumentException("Invalid HP range for " + name);
            }
        }
    }
}

package creature;

import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Collections;
import java.util.List;
import java.util.Locale;
import java.util.Objects;

/**
 * Data object shared with the battle arena, store, and CatDex scene.
 * Ability IDs are intentionally stored as strings; the battle arena owns
 * the combat behavior associated with each ID.
 */
public final class Cat {
    private int id;
    private String name;
    private String type;
    private int hp;
    private List<String> abilityIds;
    private boolean playerCat;
    private LocalDateTime createdAt;

    public Cat(
            String name,
            String type,
            int hp,
            List<String> abilityIds,
            boolean playerCat
    ) {
        this(0, name, type, hp, abilityIds, playerCat, LocalDateTime.now());
    }

    public Cat(
            int id,
            String name,
            String type,
            int hp,
            List<String> abilityIds,
            boolean playerCat,
            LocalDateTime createdAt
    ) {
        this.id = id;
        setName(name);
        setType(type);
        setHp(hp);
        setAbilityIds(abilityIds);
        this.playerCat = playerCat;
        this.createdAt = Objects.requireNonNull(createdAt, "createdAt cannot be null");
    }

    public int getId() {
        return id;
    }

    public void setId(int id) {
        if (id < 0) {
            throw new IllegalArgumentException("id cannot be negative");
        }
        this.id = id;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        if (name == null || name.isBlank()) {
            throw new IllegalArgumentException("Cat name cannot be blank");
        }
        this.name = name.trim();
    }

    public String getType() {
        return type;
    }

    public void setType(String type) {
        if (type == null || type.isBlank()) {
            throw new IllegalArgumentException("Cat type cannot be blank");
        }
        this.type = type.trim();
    }

    public int getHp() {
        return hp;
    }

    public void setHp(int hp) {
        if (hp <= 0) {
            throw new IllegalArgumentException("HP must be greater than zero");
        }
        this.hp = hp;
    }

    public List<String> getAbilityIds() {
        return List.copyOf(abilityIds);
    }

    public void setAbilityIds(List<String> abilityIds) {
        if (abilityIds == null || abilityIds.isEmpty() || abilityIds.size() > 4) {
            throw new IllegalArgumentException("A cat must have between 1 and 4 abilities");
        }

        List<String> cleaned = abilityIds.stream()
                .map(id -> Objects.requireNonNull(id, "Ability ID cannot be null").trim())
                .map(id -> id.toUpperCase(Locale.ROOT))
                .toList();

        if (cleaned.stream().anyMatch(String::isBlank)) {
            throw new IllegalArgumentException("Ability ID cannot be blank");
        }
        if (cleaned.stream().anyMatch(id -> id.contains(";"))) {
            throw new IllegalArgumentException("Ability IDs cannot contain semicolons");
        }
        if (cleaned.stream().distinct().count() != cleaned.size()) {
            throw new IllegalArgumentException("A cat cannot have duplicate abilities");
        }

        this.abilityIds = new ArrayList<>(cleaned);
    }

    public boolean isPlayerCat() {
        return playerCat;
    }

    public void setPlayerCat(boolean playerCat) {
        this.playerCat = playerCat;
    }

    public LocalDateTime getCreatedAt() {
        return createdAt;
    }

    public void setCreatedAt(LocalDateTime createdAt) {
        this.createdAt = Objects.requireNonNull(createdAt, "createdAt cannot be null");
    }

    /**
     * Canonical key used to prevent two stored cats from having the exact
     * same generated attributes. Ability order does not affect uniqueness.
     */
    public String generationKey() {
        List<String> sorted = new ArrayList<>(abilityIds);
        Collections.sort(sorted);

        return String.join(
                "|",
                name.toUpperCase(Locale.ROOT),
                type.toUpperCase(Locale.ROOT),
                Integer.toString(hp),
                String.join(";", sorted),
                Boolean.toString(playerCat)
        );
    }

    @Override
    public String toString() {
        String owner = playerCat ? "PC" : "Opponent";
        return "%s #%d - %s (%s, %d HP)".formatted(owner, id, name, type, hp);
    }
}

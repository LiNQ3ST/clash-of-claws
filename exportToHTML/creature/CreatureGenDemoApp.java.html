package creature;

import javafx.application.Application;
import javafx.fxml.FXMLLoader;
import javafx.scene.Scene;
import javafx.stage.Stage;

import java.io.IOException;

/** Standalone launcher for testing the slice before it is wired into the team scene factory. */
public final class CreatureGenDemoApp extends Application {
    @Override
    public void start(Stage stage) throws IOException {
        FXMLLoader loader = new FXMLLoader(
                CreatureGenDemoApp.class.getResource(
                        "/com/clashofcats/creaturegen/view/creature-roster.fxml"
                )
        );

        Scene scene = new Scene(loader.load(), 900, 600);
        scene.getStylesheets().add(
                CreatureGenDemoApp.class.getResource(
                        "/com/clashofcats/creaturegen/view/creature-roster.css"
                ).toExternalForm()
        );

        stage.setTitle("Clash of Cats - Creature Roster");
        stage.setScene(scene);
        stage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}

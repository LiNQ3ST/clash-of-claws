CatDAO.java 
package creature;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.sql.SQLException;
import java.sql.Statement;
import java.time.LocalDateTime;
import java.util.ArrayList;
import java.util.Arrays;
import java.util.List;
import java.util.Objects;
import java.util.Optional;

/**
 * SQLite DAO providing the required create, read, update, and delete methods.
 * Pass the same JDBC URL used by the team's shared database.
 */
public final class CatDAO {
    private static final String CREATE_TABLE_SQL = """ 
            CREATE TABLE IF NOT EXISTS cats ( 
                cat_id INTEGER PRIMARY KEY AUTOINCREMENT, 
                name TEXT NOT NULL, 
                type TEXT NOT NULL, 
                hp INTEGER NOT NULL CHECK (hp > 0), 
                abilities TEXT NOT NULL, 
                is_player_cat INTEGER NOT NULL DEFAULT 0 CHECK (is_player_cat IN (0, 1)), 
                generation_key TEXT NOT NULL UNIQUE, 
                created_at TEXT NOT NULL 
            ) 
            """;

    private final String jdbcUrl;

    public CatDAO(String jdbcUrl) {
        if (jdbcUrl == null || !jdbcUrl.startsWith("jdbc:sqlite:")) {
            throw new IllegalArgumentException("Expected a SQLite JDBC URL");
        }
        this.jdbcUrl = jdbcUrl;
    }

    public void initializeSchema() {
        try (Connection connection = openConnection();
             Statement statement = connection.createStatement()) {
            statement.execute(CREATE_TABLE_SQL);
        } catch (SQLException exception) {
            throw databaseFailure("initialize the cats table", exception);
        }
    }

    /** Create */
    public Cat insert(Cat cat) {
        Objects.requireNonNull(cat, "cat cannot be null");

        String sql = """ 
                INSERT INTO cats 
                    (name, type, hp, abilities, is_player_cat, generation_key, created_at) 
                VALUES (?, ?, ?, ?, ?, ?, ?) 
                """;

        try (Connection connection = openConnection();
             PreparedStatement statement = connection.prepareStatement(
                     sql,
                     Statement.RETURN_GENERATED_KEYS
             )) {
            bindCat(statement, cat);
            statement.executeUpdate();

            try (ResultSet keys = statement.getGeneratedKeys()) {
                if (!keys.next()) {
                    throw new SQLException("SQLite did not return a generated cat ID");
                }
                cat.setId(keys.getInt(1));
                return cat;
            }
        } catch (SQLException exception) {
            if (isUniqueConstraintFailure(exception)) {
                throw new DuplicateCatException("That exact cat is already in the CatDex", exception);
            }
            throw databaseFailure("insert the cat", exception);
        }
    }

    /** Read one */
    public Optional<Cat> findById(int catId) {
        String sql = "SELECT * FROM cats WHERE cat_id = ?";

        try (Connection connection = openConnection();
             PreparedStatement statement = connection.prepareStatement(sql)) {
            statement.setInt(1, catId);

            try (ResultSet resultSet = statement.executeQuery()) {
                return resultSet.next()
                        ? Optional.of(mapCat(resultSet))
                        : Optional.empty();
            }
        } catch (SQLException exception) {
            throw databaseFailure("read cat " + catId, exception);
        }
    }

    /** Read all */
    public List<Cat> findAll() {
        String sql = "SELECT * FROM cats ORDER BY cat_id";
        List<Cat> cats = new ArrayList<>();

        try (Connection connection = openConnection();
             PreparedStatement statement = connection.prepareStatement(sql);
             ResultSet resultSet = statement.executeQuery()) {
            while (resultSet.next()) {
                cats.add(mapCat(resultSet));
            }
            return cats;
        } catch (SQLException exception) {
            throw databaseFailure("read the CatDex", exception);
        }
    }

    /** Update */
    public boolean update(Cat cat) {
        Objects.requireNonNull(cat, "cat cannot be null");
        if (cat.getId() <= 0) {
            throw new IllegalArgumentException("Cannot update a cat without a database ID");
        }

        String sql = """ 
                UPDATE cats 
                SET name = ?, 
                    type = ?, 
                    hp = ?, 
                    abilities = ?, 
                    is_player_cat = ?, 
                    generation_key = ?, 
                    created_at = ? 
                WHERE cat_id = ? 
                """;

        try (Connection connection = openConnection();
             PreparedStatement statement = connection.prepareStatement(sql)) {
            bindCat(statement, cat);
            statement.setInt(8, cat.getId());
            return statement.executeUpdate() == 1;
        } catch (SQLException exception) {
            if (isUniqueConstraintFailure(exception)) {
                throw new DuplicateCatException("The update would duplicate another CatDex entry", exception);
            }
            throw databaseFailure("update cat " + cat.getId(), exception);
        }
    }

    /** Delete */
    public boolean delete(int catId) {
        String sql = "DELETE FROM cats WHERE cat_id = ?";

        try (Connection connection = openConnection();
             PreparedStatement statement = connection.prepareStatement(sql)) {
            statement.setInt(1, catId);
            return statement.executeUpdate() == 1;
        } catch (SQLException exception) {
            throw databaseFailure("delete cat " + catId, exception);
        }
    }

    public boolean exists(Cat cat) {
        String sql = "SELECT 1 FROM cats WHERE generation_key = ? LIMIT 1";

        try (Connection connection = openConnection();
             PreparedStatement statement = connection.prepareStatement(sql)) {
            statement.setString(1, cat.generationKey());
            try (ResultSet resultSet = statement.executeQuery()) {
                return resultSet.next();
            }
        } catch (SQLException exception) {
            throw databaseFailure("check for a duplicate cat", exception);
        }
    }

    public long countOpponentCats() {
        String sql = "SELECT COUNT(*) FROM cats WHERE is_player_cat = 0";

        try (Connection connection = openConnection();
             PreparedStatement statement = connection.prepareStatement(sql);
             ResultSet resultSet = statement.executeQuery()) {
            return resultSet.next() ? resultSet.getLong(1) : 0L;
        } catch (SQLException exception) {
            throw databaseFailure("count opponent cats", exception);
        }
    }

    private Connection openConnection() throws SQLException {
        return DriverManager.getConnection(jdbcUrl);
    }

    private static void bindCat(PreparedStatement statement, Cat cat) throws SQLException {
        statement.setString(1, cat.getName());
        statement.setString(2, cat.getType());
        statement.setInt(3, cat.getHp());
        statement.setString(4, String.join(";", cat.getAbilityIds()));
        statement.setBoolean(5, cat.isPlayerCat());
        statement.setString(6, cat.generationKey());
        statement.setString(7, cat.getCreatedAt().toString());
    }

    private static Cat mapCat(ResultSet resultSet) throws SQLException {
        String storedAbilities = resultSet.getString("abilities");
        List<String> abilityIds = storedAbilities.isBlank()
                ? List.of()
                : Arrays.asList(storedAbilities.split(";"));

        return new Cat(
                resultSet.getInt("cat_id"),
                resultSet.getString("name"),
                resultSet.getString("type"),
                resultSet.getInt("hp"),
                abilityIds,
                resultSet.getBoolean("is_player_cat"),
                LocalDateTime.parse(resultSet.getString("created_at"))
        );
    }

    private static boolean isUniqueConstraintFailure(SQLException exception) {
        String message = exception.getMessage();
        return message != null && message.toLowerCase().contains("unique");
    }

    private static IllegalStateException databaseFailure(String action, SQLException exception) {
        return new IllegalStateException("Could not " + action + ": " + exception.getMessage(), exception);
    }

    public static final class DuplicateCatException extends IllegalStateException {
        public DuplicateCatException(String message, Throwable cause) {
            super(message, cause);
        }
    }
}

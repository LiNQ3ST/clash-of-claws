package creature;

import creature.CatGenerator.Cat;

import java.io.BufferedReader;
import java.io.BufferedWriter;
import java.io.IOException;
import java.nio.charset.StandardCharsets;
import java.nio.file.Files;
import java.nio.file.Path;
import java.nio.file.StandardOpenOption;
import java.util.ArrayList;
import java.util.HashSet;
import java.util.List;
import java.util.Set;

public class CatDex {

    private static final String HEADER =
            "name,type,hp,abilities,isPlayerCat";

    private final Path catDexFile;
    private final List<Cat> cats;
    private final Set<String> catSignatures;

    public CatDex(Path catDexFile) {
        this.catDexFile = catDexFile;
        this.cats = new ArrayList<>();
        this.catSignatures = new HashSet<>();

        initializeFile();
        load();
    }

    /**
     * Stores a cat immediately after creation.
     */
    public synchronized void save(Cat cat) {
        if (cat == null) {
            throw new IllegalArgumentException(
                    "Cannot save a null cat."
            );
        }

        if (contains(cat)) {
            return;
        }

        try (BufferedWriter writer = Files.newBufferedWriter(
                catDexFile,
                StandardCharsets.UTF_8,
                StandardOpenOption.CREATE,
                StandardOpenOption.APPEND
        )) {
            writer.write(toCsvLine(cat));
            writer.newLine();

            cats.add(cat);
            catSignatures.add(cat.signature());

        } catch (IOException exception) {
            throw new IllegalStateException(
                    "Could not save the cat to the CatDex.",
                    exception
            );
        }
    }

    public boolean contains(Cat cat) {
        return catSignatures.contains(cat.signature());
    }

    public List<Cat> getAllCats() {
        return List.copyOf(cats);
    }

    public int size() {
        return cats.size();
    }

    public boolean isEmpty() {
        return cats.isEmpty();
    }

    private void initializeFile() {
        try {
            Path parent = catDexFile.getParent();

            if (parent != null) {
                Files.createDirectories(parent);
            }

            if (Files.notExists(catDexFile)) {
                Files.writeString(
                        catDexFile,
                        HEADER + System.lineSeparator(),
                        StandardCharsets.UTF_8,
                        StandardOpenOption.CREATE_NEW
                );
            }

        } catch (IOException exception) {
            throw new IllegalStateException(
                    "Could not initialize the CatDex file.",
                    exception
            );
        }
    }

    private void load() {
        cats.clear();
        catSignatures.clear();

        try (BufferedReader reader = Files.newBufferedReader(
                catDexFile,
                StandardCharsets.UTF_8
        )) {
            String line;
            boolean firstLine = true;

            while ((line = reader.readLine()) != null) {
                if (firstLine) {
                    firstLine = false;
                    continue;
                }

                if (line.isBlank()) {
                    continue;
                }

                Cat cat = fromCsvLine(line);

                cats.add(cat);
                catSignatures.add(cat.signature());
            }

        } catch (IOException exception) {
            throw new IllegalStateException(
                    "Could not load the CatDex.",
                    exception
            );
        }
    }

    private String toCsvLine(Cat cat) {
        String abilityText =
                String.join(";", cat.abilities());

        return String.join(
                ",",
                escapeCsv(cat.name()),
                escapeCsv(cat.type()),
                Integer.toString(cat.hp()),
                escapeCsv(abilityText),
                Boolean.toString(cat.playerCat())
        );
    }

    private Cat fromCsvLine(String line) {
        List<String> values = parseCsvLine(line);

        if (values.size() != 5) {
            throw new IllegalStateException(
                    "Invalid CatDex entry: " + line
            );
        }

        String name = values.get(0);
        String type = values.get(1);
        int hp = Integer.parseInt(values.get(2));

        List<String> abilities = List.of(
                values.get(3).split(";")
        );

        boolean playerCat =
                Boolean.parseBoolean(values.get(4));

        return new Cat(
                name,
                type,
                hp,
                abilities,
                playerCat
        );
    }

    private String escapeCsv(String value) {
        String escaped = value.replace("\"", "\"\"");

        if (escaped.contains(",")
                || escaped.contains("\"")
                || escaped.contains("\n")) {

            return "\"" + escaped + "\"";
        }

        return escaped;
    }

    private List<String> parseCsvLine(String line) {
        List<String> values = new ArrayList<>();
        StringBuilder currentValue = new StringBuilder();

        boolean insideQuotes = false;

        for (int index = 0; index < line.length(); index++) {
            char character = line.charAt(index);

            if (character == '"') {
                if (insideQuotes
                        && index + 1 < line.length()
                        && line.charAt(index + 1) == '"') {

                    currentValue.append('"');
                    index++;
                } else {
                    insideQuotes = !insideQuotes;
                }
            } else if (character == ',' && !insideQuotes) {
                values.add(currentValue.toString());
                currentValue.setLength(0);
            } else {
                currentValue.append(character);
            }
        }

        values.add(currentValue.toString());
        return values;
    }
}
